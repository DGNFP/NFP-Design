            function showStyle(styleNumber) {
                // 모든 버튼의 active 클래스 제거
                document.querySelectorAll('.control-button').forEach(btn => {
                    btn.classList.remove('active');
                });
                
                // 클릭된 버튼에 active 클래스 추가
                event.target.classList.add('active');
                
                // 모든 스타일 숨기기
                document.querySelectorAll('.phone-mockup').forEach(phone => {
                    phone.style.display = 'none';
                });
                
                // 선택된 스타일 보이기
                document.getElementById('style' + styleNumber).style.display = 'block';
            }

            // 페이지 로드 시 첫 번째 스타일 표시
            document.addEventListener('DOMContentLoaded', function() {
                showStyle(1);
            });

            // 자동 순환 (선택사항)
            let currentStyle = 1;
            setInterval(() => {
                currentStyle = currentStyle === 3 ? 1 : currentStyle + 1;
                showStyle(currentStyle);
                
                // 버튼 상태도 업데이트
                document.querySelectorAll('.control-button').forEach((btn, index) => {
                    btn.classList.toggle('active', index + 1 === currentStyle);
                });
            }, 5000); // 5초마다 변경
