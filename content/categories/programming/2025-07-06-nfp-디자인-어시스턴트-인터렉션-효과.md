            // 1. 파티클 폭발 효과
            function triggerParticles() {
                const card = document.getElementById('particle-card');
                const rect = card.getBoundingClientRect();
                const centerX = rect.left + rect.width / 2;
                const centerY = rect.top + rect.height / 2;
                const cardColor = '#01FF75';

                // 20개의 파티클 생성
                for (let i = 0; i < 20; i++) {
                    const particle = document.createElement('div');
                    particle.className = 'particle';
                    particle.style.backgroundColor = cardColor;
                    particle.style.left = centerX + 'px';
                    particle.style.top = centerY + 'px';
                    
                    document.body.appendChild(particle);

                    // 랜덤한 방향으로 파티클 이동
                    const angle = (Math.PI * 2 * i) / 20;
                    const velocity = 100 + Math.random() * 100;
                    const deltaX = Math.cos(angle) * velocity;
                    const deltaY = Math.sin(angle) * velocity;

                    // 파티클 애니메이션
                    particle.animate([
                        { transform: 'translate(0, 0) scale(1)', opacity: 1 },
                        { transform: `translate(${deltaX}px, ${deltaY}px) scale(0)`, opacity: 0 }
                    ], {
                        duration: 1000,
                        easing: 'ease-out'
                    }).onfinish = () => particle.remove();
                }
            }

            // 2. 하늘에서 떨어지는 효과
            function triggerFalling() {
                const card = document.getElementById('falling-card');
                card.classList.remove('falling-card');
                
                // 약간의 지연 후 애니메이션 시작
                setTimeout(() => {
                    card.classList.add('falling-card');
                    
                    // 애니메이션 완료 후 클래스 제거
                    setTimeout(() => {
                        card.classList.remove('falling-card');
                    }, 1000);
                }, 50);
            }

            // 3. 슬롯머신 효과
            function triggerSlotMachine() {
                const card = document.getElementById('slot-card');
                const colors = ['#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4', '#FECA57', '#FF9FF3'];
                
                // 슬롯머신 애니메이션 시작
                card.classList.add('slot-spinning');
                
                // 3초 후 랜덤 색상에서 멈춤
                setTimeout(() => {
                    card.classList.remove('slot-spinning');
                    const finalColor = colors[Math.floor(Math.random() * colors.length)];
                    card.style.backgroundColor = finalColor;
                    card.textContent = finalColor;
                    
                    // 약간의 펄스 효과
                    card.animate([
                        { transform: 'scale(1)' },
                        { transform: 'scale(1.1)' },
                        { transform: 'scale(1)' }
                    ], {
                        duration: 300,
                        easing: 'ease-out'
                    });
                }, 3000);
            }
        </script>
